	\section*{Exercise 6.2.1 - Convert into first-order form}
	\begin{align}
		&\frac{d^2x}{dt^2} - x \frac{dx}{dt} + x^2 = 0, \quad y = \frac{dx}{dt} \\
		&\frac{dy}{dt} - xy + x^2 = 0 \\
		&\frac{dy}{dt} = xy - x^2
	\end{align}
	
	\section*{Exercise 6.2.2 - Convert into autonomous first-order form}
	\begin{align}
		&\frac{d^2x}{dt^2} - a\cos(bt) = 0, \quad \frac{d\tau}{dt}=1, \quad y = \frac{dx}{dt} \\
		&\frac{dy}{dt} - a\cos(b\tau) = 0, \quad \tau(0) = 0, \quad \tau(t)=t
	\end{align}
